3. Update `processAudio` to include report generation after transcription:
   ```typescript
   const processAudio = async () => {
     try {
       setAppState('PROCESSING');

       // Step 1: Transcribe
       setProcessingStep('Transcribing audio...');
       const transcript = await transcribeChunks(audioChunks, (current, total) => {
         setProcessingStep(`Transcribing chunk ${current}/${total}...`);
       });
       setTranscript(transcript);

       // Step 2: Generate reports
       setProcessingStep('Generating report...');
       const result = await generateReport(transcript, currentLanguage);
       setAllResults(result);

       setAppState('RESULTS');
     } catch (error: any) {
       Alert.alert('Processing Error', error.message || 'Failed to process audio.');
       setAppState('READY');
     }
   };
   ```
